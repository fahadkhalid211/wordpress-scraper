<script>
(function() {
  const API_URL = "https://glymph-chatbot.onrender.com/api/chat";

  const launcher = document.getElementById('glymph-chat-launcher');
  const win = document.getElementById('glymph-chat-window');
  const messages = document.getElementById('glymph-chat-messages');
  const input = document.getElementById('glymph-chat-input');
  const sendBtn = document.getElementById('glymph-chat-send');

  launcher.addEventListener('click', () => win.classList.toggle('open'));

  function addMessage(text, sender) {
    const div = document.createElement('div');
    div.className = 'glymph-msg ' + sender;
    div.textContent = text;
    messages.appendChild(div);
    messages.scrollTop = messages.scrollHeight;
    return div;
  }

  async function sendMessage() {
    const text = input.value.trim();
    if (!text) return;
    addMessage(text, 'user');
    input.value = '';
    const typingEl = addMessage('Typing...', 'typing');

    try {
      const res = await fetch(API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ message: text })
      });
      const data = await res.json();
      typingEl.remove();
      addMessage(data.answer || "Sorry, I couldn't find an answer.", 'bot');
    } catch (e) {
      typingEl.remove();
      addMessage("Connection error. Please try again.", 'bot');
    }
  }

  sendBtn.addEventListener('click', sendMessage);
  input.addEventListener('keypress', (e) => { if (e.key === 'Enter') sendMessage(); });
})();
</script>
